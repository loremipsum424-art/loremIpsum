def smote_balance(X_df, y_series, target_counts=None, k_neighbors_max=5):
    counts = y_series.value_counts()

    if target_counts is None:
        strategy = "auto"
        min_count = counts.min()
    else:
        strategy = {cls: target for cls, target in target_counts.items() if target > counts.get(cls, 0)}
        if not strategy:
            return X_df.reset_index(drop=True), y_series.reset_index(drop=True)
        min_count = min([counts.get(cls, 0) for cls in strategy.keys()])

    if min_count < 2:
        print("SMOTE not possible for smallest class; using RandomOverSampler fallback.")
        ros = RandomOverSampler(sampling_strategy=strategy, random_state=RANDOM_STATE)
        X_res, y_res = ros.fit_resample(X_df, y_series)
        return pd.DataFrame(X_res, columns=X_df.columns), pd.Series(y_res, name=y_series.name)

    k = min(k_neighbors_max, min_count - 1)
    print(f"Applying SMOTE with k_neighbors={k}")
    sm = SMOTE(sampling_strategy=strategy, k_neighbors=k, random_state=RANDOM_STATE)
    X_res, y_res = sm.fit_resample(X_df, y_series)
    return pd.DataFrame(X_res, columns=X_df.columns), pd.Series(y_res, name=y_series.name)

@tf.keras.utils.register_keras_serializable()
def focal_loss(y_true, y_pred, gamma=2.0, alpha=0.25):
    y_pred = tf.clip_by_value(y_pred, tf.keras.backend.epsilon(), 1.0 - tf.keras.backend.epsilon())
    ce = -y_true * tf.math.log(y_pred)
    w = alpha * tf.pow(1.0 - y_pred, gamma)
    return tf.reduce_sum(w * ce, axis=1)

@tf.keras.utils.register_keras_serializable()
class WarmupCosine(tf.keras.optimizers.schedules.LearningRateSchedule):
    def __init__(self, base_lr, warmup_steps, total_steps, **kwargs):
        super().__init__()
        self.base_lr = base_lr
        self.warmup_steps = warmup_steps
        self.total_steps = total_steps

    def __call__(self, step):
        step = tf.cast(step, tf.float32)
        warm = tf.cast(self.warmup_steps, tf.float32)
        total = tf.cast(self.total_steps, tf.float32)
        return tf.where(
            step < warm,
            self.base_lr * (step / tf.maximum(1.0, warm)),
            0.5 * self.base_lr * (1.0 + tf.cos(np.pi * (step - warm) / tf.maximum(1.0, total - warm))),
        )

    def get_config(self):
        return {"base_lr": self.base_lr, "warmup_steps": self.warmup_steps, "total_steps": self.total_steps}

@tf.keras.utils.register_keras_serializable()
class NumericFeatureTokenizer(layers.Layer):
    def __init__(self, num_features, d_model, **kwargs):
        super().__init__(**kwargs)
        self.num_features = num_features
        self.d_model = d_model

    def build(self, input_shape):
        self.weight = self.add_weight(shape=(self.num_features, self.d_model), initializer="glorot_uniform", trainable=True)
        self.bias = self.add_weight(shape=(self.num_features, self.d_model), initializer="zeros", trainable=True)

    def call(self, inputs):
        x = tf.expand_dims(inputs, axis=-1)
        return x * self.weight + self.bias

    def get_config(self):
        cfg = super().get_config()
        cfg.update({"num_features": self.num_features, "d_model": self.d_model})
        return cfg

@tf.keras.utils.register_keras_serializable()
class TransformerBlock(layers.Layer):
    def __init__(self, d_model, num_heads, ff_dim, dropout=0.1, **kwargs):
        super().__init__(**kwargs)
        self.d_model = d_model
        self.num_heads = num_heads
        self.ff_dim = ff_dim
        self.dropout_rate = dropout
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=max(1, d_model // num_heads))
        self.ffn = tf.keras.Sequential([
            layers.Dense(ff_dim, activation="relu"),
            layers.Dropout(dropout),
            layers.Dense(d_model),
        ])
        self.norm1 = layers.LayerNormalization(epsilon=1e-6)
        self.norm2 = layers.LayerNormalization(epsilon=1e-6)
        self.drop1 = layers.Dropout(dropout)
        self.drop2 = layers.Dropout(dropout)

    def call(self, inputs, training=False):
        x = self.norm1(inputs)
        attn = self.att(x, x, training=training)
        attn = self.drop1(attn, training=training)
        x = inputs + attn
        y = self.norm2(x)
        y = self.ffn(y, training=training)
        y = self.drop2(y, training=training)
        return x + y

    def get_config(self):
        cfg = super().get_config()
        cfg.update({"d_model": self.d_model, "num_heads": self.num_heads, "ff_dim": self.ff_dim, "dropout": self.dropout_rate})
        return cfg

# =========================
# Model
# =========================
def build_model(input_dim, num_classes, d_model, num_heads, ff_dim, num_blocks, dropout):
    inp = layers.Input(shape=(input_dim,))
    x = NumericFeatureTokenizer(input_dim, d_model)(inp)
    x = layers.LayerNormalization(epsilon=1e-6)(x)
    pos = layers.Embedding(input_dim=input_dim, output_dim=d_model)(tf.range(0, input_dim))
    x = x + pos
    x = layers.SpatialDropout1D(dropout)(x)

    for _ in range(num_blocks):
        x = TransformerBlock(d_model, num_heads, ff_dim, dropout=dropout)(x)

    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dense(128, activation="relu", kernel_constraint=constraints.MaxNorm(MAX_NORM_VALUE))(x)
    x = layers.Dropout(dropout)(x)
    x = layers.Dense(64, activation="relu", kernel_constraint=constraints.MaxNorm(MAX_NORM_VALUE))(x)
    x = layers.Dropout(dropout)(x)
    out = layers.Dense(num_classes, activation="softmax")(x)
    return Model(inp, out)

def make_optimizer(train_ds, epochs, lr, weight_decay):
    steps_per_epoch = int(tf.data.experimental.cardinality(train_ds).numpy())
    total_steps = max(1, steps_per_epoch * epochs)
    warmup_steps = max(1, steps_per_epoch * WARMUP_EPOCHS)
    return tf.keras.optimizers.AdamW(
        learning_rate=WarmupCosine(lr, warmup_steps, total_steps),
        weight_decay=weight_decay,
    )

def evaluate_model(model, ds, y_true, labels, prefix):
    prob = model.predict(ds, verbose=0)
    pred = np.argmax(prob, axis=1)
    acc = accuracy_score(y_true, pred)
    mf1 = f1_score(y_true, pred, average="macro", zero_division=0)
    rep = classification_report(y_true, pred, target_names=labels, digits=4, zero_division=0)
    print(f"\n===== {prefix} =====")
    print("Accuracy:", round(acc, 4))
    print("Macro F1:", round(mf1, 4))
    print(rep)

    cm = confusion_matrix(y_true, pred, labels=range(len(labels)))
    print("Per-class TPR/FPR:")
    for i, cls in enumerate(labels):
        tp = cm[i, i]
        fn = cm[i, :].sum() - tp
        fp = cm[:, i].sum() - tp
        tn = cm.sum() - tp - fn - fp
        tpr = tp / (tp + fn) if tp + fn else 0.0
        fpr = fp / (fp + tn) if fp + tn else 0.0
        print(f"  {cls}: TPR={tpr:.4f}, FPR={fpr:.4f}")

    with open(os.path.join(OUTPUT_DIR, f"{prefix.lower()}_report.txt"), "w") as f:
        f.write(f"Accuracy: {acc:.4f}\n")
        f.write(f"Macro F1: {mf1:.4f}\n\n")
        f.write(rep)
    return pred, prob

def plot_history(history, title, fname):
    plt.figure(figsize=(12, 5))
    plt.subplot(1, 2, 1)
    plt.plot(history.history["accuracy"], label="Train")
    plt.plot(history.history["val_accuracy"], label="Val")
    plt.title(f"{title} Accuracy")
    plt.legend()
    plt.subplot(1, 2, 2)
    plt.plot(history.history["loss"], label="Train")
    plt.plot(history.history["val_loss"], label="Val")
    plt.title(f"{title} Loss")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_DIR, f"{fname}.png"), dpi=200)
    plt.show()
