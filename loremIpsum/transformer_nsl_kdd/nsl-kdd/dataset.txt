"""
NSL-KDD Binary + Multiclass Transformer (One-Hot + MI Selection + SMOTE)
==========================================================================
CTGAN replaced with SMOTE in this version. Reason: your kernel crash was
almost certainly TensorFlow + PyTorch (which `ctgan` uses internally)
fighting over GPU memory/CUDA context in the same process — a common,
often silent (no Python traceback) failure mode. SMOTE is pure
scikit-learn/NumPy, so this removes that entire class of crash and is
also far faster (SMOTE is near-instant; CTGAN was training a GAN per
class for 300 epochs).

Tradeoff to note explicitly for your paper: SMOTE generates synthetic
samples via linear interpolation between a minority sample and its
k-nearest same-class neighbors, which is a much simpler mechanism than
CTGAN's learned generative distribution. With U2R having only a few dozen
real training samples, the neighbor pool SMOTE interpolates within is
small, so synthetic U2R samples cluster close to the few real ones —
state this as a limitation rather than presenting it as equivalent to
CTGAN.

All other fixes from the previous version are retained:
  - Complete, asserted attack_map ('spy' -> R2L).
  - COLS includes 'difficulty' (the earlier missing-column bug that
    caused pandas to silently shift every column name by one).
  - One-hot vocabulary built from TRAIN ONLY, test reindexed onto it.
  - log1p on skewed numeric columns before scaling.
  - Zero-variance / high-correlation feature dropping (train-derived).
  - MI feature selection unions top features from BOTH binary and
    multiclass scoring.
  - NumericFeatureTokenizer (per-feature weight+bias, not a shared Dense).
  - WarmupCosine.get_config() present (required for ModelCheckpoint save).
  - ReduceLROnPlateau excluded (incompatible with a LearningRateSchedule
    optimizer).
  - Per-class FPR/TPR reporting.

pip install tensorflow scikit-learn pandas numpy matplotlib seaborn joblib imbalanced-learn
"""

# Balance with SMOTE
# =========================
def smote_balance(X_df, y_series, target_counts=None, k_neighbors_max=SMOTE_K_NEIGHBORS_MAX):
    """
    If target_counts is None: SMOTE with sampling_strategy='auto' (every
    minority class oversampled to match the majority class count).
    If target_counts is a dict {class_label: desired_count}: only those
    classes are oversampled, to the given target (classes not listed are
    left untouched).

    k_neighbors is reduced automatically to stay valid for the smallest
    class being oversampled. If that class has fewer than 2 real samples,
    SMOTE cannot interpolate at all and RandomOverSampler (exact
    duplication) is used instead as a safe fallback.
    """
    if not USE_SMOTE:
        return X_df.reset_index(drop=True), y_series.reset_index(drop=True)

    counts = y_series.value_counts()

    if target_counts is None:
        sampling_strategy = 'auto'
        relevant_counts = counts
    else:
        sampling_strategy = {
            cls: target for cls, target in target_counts.items()
            if target > counts.get(cls, 0)
        }
        if not sampling_strategy:
            print("No classes need oversampling under the given targets; skipping SMOTE.")
            return X_df.reset_index(drop=True), y_series.reset_index(drop=True)
        relevant_counts = {cls: counts.get(cls, 0) for cls in sampling_strategy}

    min_relevant_count = min(relevant_counts.values()) if isinstance(relevant_counts, dict) else relevant_counts.min()

    if min_relevant_count < 2:
        print(f"A target class has fewer than 2 real samples ({min_relevant_count}); "
              f"SMOTE cannot interpolate. Falling back to RandomOverSampler (exact "
              f"duplication, not synthetic) for this call.")
        ros = RandomOverSampler(sampling_strategy=sampling_strategy, random_state=RANDOM_STATE)
        X_res, y_res = ros.fit_resample(X_df, y_series)
        return (pd.DataFrame(X_res, columns=X_df.columns).reset_index(drop=True),
                pd.Series(y_res, name=y_series.name).reset_index(drop=True))

    k = min(k_neighbors_max, min_relevant_count - 1)
    sm = SMOTE(sampling_strategy=sampling_strategy, k_neighbors=k, random_state=RANDOM_STATE)
    X_res, y_res = sm.fit_resample(X_df, y_series)
    return (pd.DataFrame(X_res, columns=X_df.columns).reset_index(drop=True),
            pd.Series(y_res, name=y_series.name).reset_index(drop=True))

def make_ds(X, y, training=False):

@tf.keras.utils.register_keras_serializable()
def focal_loss(y_true, y_pred, gamma=2.0, alpha=0.25):
    y_pred = tf.clip_by_value(y_pred, tf.keras.backend.epsilon(), 1.0 - tf.keras.backend.epsilon())
    ce = -y_true * tf.math.log(y_pred)
    w = alpha * tf.pow(1.0 - y_pred, gamma)
    return tf.reduce_sum(w * ce, axis=1)


@tf.keras.utils.register_keras_serializable()
class WarmupCosine(tf.keras.optimizers.schedules.LearningRateSchedule):
    def __init__(self, base_lr, warmup_steps, total_steps, **kwargs):
        super().__init__()
        self.base_lr = base_lr
        self.warmup_steps = warmup_steps
        self.total_steps = total_steps

    def __call__(self, step):
        step = tf.cast(step, tf.float32)
        warm = tf.cast(self.warmup_steps, tf.float32)
        total = tf.cast(self.total_steps, tf.float32)
        return tf.where(
            step < warm,
            self.base_lr * (step / tf.maximum(1.0, warm)),
            0.5 * self.base_lr * (1.0 + tf.cos(np.pi * (step - warm) / tf.maximum(1.0, total - warm))),
        )

    def get_config(self):
        return {"base_lr": self.base_lr, "warmup_steps": self.warmup_steps, "total_steps": self.total_steps}


@tf.keras.utils.register_keras_serializable()
class NumericFeatureTokenizer(layers.Layer):
    def __init__(self, num_features, d_model, **kwargs):
        super().__init__(**kwargs)
        self.num_features = num_features
        self.d_model = d_model

    def build(self, input_shape):
        self.weight = self.add_weight(
            shape=(self.num_features, self.d_model), initializer="glorot_uniform",
            trainable=True, name="numeric_weight",
        )
        self.bias = self.add_weight(
            shape=(self.num_features, self.d_model), initializer="zeros",
            trainable=True, name="numeric_bias",
        )

    def call(self, inputs):
        x = tf.expand_dims(inputs, axis=-1)
        return x * self.weight + self.bias

    def get_config(self):
        config = super().get_config()
        config.update({"num_features": self.num_features, "d_model": self.d_model})
        return config


@tf.keras.utils.register_keras_serializable()
class TransformerBlock(layers.Layer):
    def __init__(self, d_model, num_heads, ff_dim, dropout=0.1, **kwargs):
        super().__init__(**kwargs)
        self.d_model = d_model
        self.num_heads = num_heads
        self.ff_dim = ff_dim
        self.dropout_rate = dropout
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=max(1, d_model // num_heads))
        self.ffn = tf.keras.Sequential([
            layers.Dense(ff_dim, activation="relu"),
            layers.Dropout(dropout),
            layers.Dense(d_model),
        ])
        self.norm1 = layers.LayerNormalization(epsilon=1e-6)
        self.norm2 = layers.LayerNormalization(epsilon=1e-6)
        self.drop1 = layers.Dropout(dropout)
        self.drop2 = layers.Dropout(dropout)

    def call(self, inputs, training=False):
        x = self.norm1(inputs)
        attn = self.att(x, x, training=training)
        attn = self.drop1(attn, training=training)
        x = inputs + attn
        y = self.norm2(x)
        y = self.ffn(y, training=training)
        y = self.drop2(y, training=training)
        return x + y

    def get_config(self):
        cfg = super().get_config()
        cfg.update({
            "d_model": self.d_model, "num_heads": self.num_heads,
            "ff_dim": self.ff_dim, "dropout": self.dropout_rate,
        })
        return cfg


class VerboseEpochLogger(tf.keras.callbacks.Callback):
    """Prints timing, current learning rate (useful for watching
    WarmupCosine actually warm up and decay), and all logged metrics at
    the start/end of every epoch — gives full visibility into training
    beyond Keras's default progress bar."""
    def __init__(self, stage_name):
        super().__init__()
        self.stage_name = stage_name
        self._epoch_start = None

    def on_train_begin(self, logs=None):
        log(f"[{self.stage_name}] Training started. "
            f"Model has {self.model.count_params():,} parameters.")

    def on_epoch_begin(self, epoch, logs=None):
        self._epoch_start = time.time()

    def on_epoch_end(self, epoch, logs=None):
        duration = time.time() - self._epoch_start
        logs = logs or {}
        try:
            lr_schedule = self.model.optimizer.learning_rate
            current_step = self.model.optimizer.iterations
            current_lr = float(lr_schedule(current_step)) if callable(lr_schedule) else float(lr_schedule)
        except Exception:
            current_lr = float('nan')
        metrics_str = ", ".join(f"{k}={v:.4f}" for k, v in logs.items())
        log(f"[{self.stage_name}] Epoch {epoch + 1} finished in {duration:.1f}s "
            f"| lr={current_lr:.2e} | {metrics_str}")

    def on_train_end(self, logs=None):
        log(f"[{self.stage_name}] Training finished.")


def build_model(input_dim, num_classes):
    inp = layers.Input(shape=(input_dim,))
    x = NumericFeatureTokenizer(input_dim, D_MODEL)(inp)
    x = layers.LayerNormalization(epsilon=1e-6)(x)

    positions = tf.range(start=0, limit=input_dim, delta=1)
    pos_emb = layers.Embedding(input_dim=input_dim, output_dim=D_MODEL)
    x = x + pos_emb(positions)
    x = layers.SpatialDropout1D(DROPOUT)(x)

    for _ in range(NUM_TRANSFORMER_BLOCKS):
        x = TransformerBlock(D_MODEL, NUM_HEADS, FF_DIM, dropout=DROPOUT)(x)

    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dense(128, activation="relu", kernel_constraint=constraints.MaxNorm(MAX_NORM_VALUE))(x)
    x = layers.Dropout(DROPOUT)(x)
    x = layers.Dense(64, activation="relu", kernel_constraint=constraints.MaxNorm(MAX_NORM_VALUE))(x)
    x = layers.Dropout(DROPOUT)(x)
    out = layers.Dense(num_classes, activation="softmax")(x)
    return Model(inp, out)


def make_optimizer(train_ds, epochs):
    steps_per_epoch = tf.data.experimental.cardinality(train_ds).numpy()
    total_steps = max(1, steps_per_epoch * epochs)
    warmup_steps = max(1, steps_per_epoch * WARMUP_EPOCHS)
    lr_schedule = WarmupCosine(BASE_LR, warmup_steps, total_steps)
    return tf.keras.optimizers.AdamW(learning_rate=lr_schedule, weight_decay=WEIGHT_DECAY)


def plot_history(history, title, fname):
    plt.figure(figsize=(12, 5))
    plt.subplot(1, 2, 1)
    plt.plot(history.history["accuracy"], label="Train")
    plt.plot(history.history["val_accuracy"], label="Val")
    plt.legend(); plt.title(f"{title} Accuracy")
    plt.subplot(1, 2, 2)
    plt.plot(history.history["loss"], label="Train")
    plt.plot(history.history["val_loss"], label="Val")
    plt.legend(); plt.title(f"{title} Loss")
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_DIR, f"{fname}.png"), dpi=200)
    plt.show()


def evaluate(model, ds, y_true, labels, prefix):
    prob = model.predict(ds, verbose=0)
    pred = np.argmax(prob, axis=1)
    acc = accuracy_score(y_true, pred)
    mf1 = f1_score(y_true, pred, average="macro", zero_division=0)
    rep = classification_report(y_true, pred, target_names=labels, digits=4, zero_division=0)
    print(f"\n===== {prefix} =====")
    print("Accuracy:", round(acc, 4))
    print("Macro F1:", round(mf1, 4))
    print(rep)

    cm = confusion_matrix(y_true, pred, labels=range(len(labels)))
    print(f"{'Class':<10}{'TPR(Recall)':<14}{'FPR':<10}{'FP':<6}{'TN':<8}")
    fpr_dict = {}
    for i, cname in enumerate(labels):
        tp = cm[i, i]
        fn = cm[i, :].sum() - tp
        fp = cm[:, i].sum() - tp
        tn = cm.sum() - tp - fn - fp
        tpr = tp / (tp + fn) if (tp + fn) > 0 else 0.0
        fpr = fp / (fp + tn) if (fp + tn) > 0 else 0.0
        fpr_dict[cname] = fpr
        print(f"{cname:<10}{tpr:<14.4f}{fpr:<10.4f}{fp:<6}{tn:<8}")
    mean_fpr = np.mean(list(fpr_dict.values()))
    print(f"\nMean per-class FPR: {mean_fpr:.4f}")

    with open(os.path.join(OUTPUT_DIR, f"{prefix.lower()}_report.txt"), "w") as f:
        f.write(f"Accuracy: {acc:.4f}\nMacro F1: {mf1:.4f}\nMean per-class FPR: {mean_fpr:.4f}\n\n")
        f.write(rep)
    return pred, prob
